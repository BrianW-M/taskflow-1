function showAuthTab(tab, el) {
  document.querySelectorAll('.auth-tab').forEach(t => t.classList.remove('active'));
  if (el) el.classList.add('active');
  document.querySelectorAll('#tab-login,#tab-signup').forEach(t => t.classList.add('hidden'));
  document.getElementById(`tab-${tab}`).classList.remove('hidden');
  hideAuthMsg();
}

async function doLogin() {
  const email = document.getElementById('login-email').value.trim();
  const pass = document.getElementById('login-password').value;
  if (!email || !pass) return showAuthError('Please enter email and password.');
  setLoading('login-btn', true);
  const { error } = await sb.auth.signInWithPassword({ email, password: pass });
  setLoading('login-btn', false);
  if (error) showAuthError(error.message);
}

async function doSignup() {
  const name = document.getElementById('signup-name').value.trim();
  const email = document.getElementById('signup-email').value.trim();
  const pass = document.getElementById('signup-password').value;
  if (!name || !email || !pass) return showAuthError('Please fill in all fields.');
  if (pass.length < 6) return showAuthError('Password must be at least 6 characters.');
  setLoading('signup-btn', true);
  const { error } = await sb.auth.signUp({
    email, password: pass,
    options: { data: { full_name: name } }
  });
  setLoading('signup-btn', false);
  if (error) showAuthError(error.message);
  else showAuthSuccess('Account created! Check your email to confirm, then sign in.');
}

// ─── PROJECTS ─────────────────────────────────────────────────────
async function loadProjects() {
  // Get projects where user is owner or member
  const { data: owned } = await sb.from('projects').select('*').eq('owner_id', currentUser.id);
  const { data: memberships } = await sb.from('project_members')
    .select('project_id, projects(*)')
    .eq('user_id', currentUser.id);

  const memberProjects = (memberships || []).map(m => m.projects).filter(Boolean);
  const allProjects = [...(owned || []), ...memberProjects].filter((p, i, arr) => arr.findIndex(x => x.id === p.id) === i);

  const nav = document.getElementById('project-list-nav');
  nav.innerHTML = '';
  if (allProjects.length === 0) {
    // Create a default project
    await createDefaultProject();
    return loadProjects();
  }

  allProjects.forEach((p, i) => {
    const div = document.createElement('div');
    div.className = 'nav-item' + (i === 0 ? ' active' : '');
    div.innerHTML = `<span class="nav-proj-dot" style="background:${p.color || '#7090f0'}"></span>${p.name}`;
    div.onclick = () => {
      document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));
      div.classList.add('active');
      loadProject(p);
    };
    nav.appendChild(div);
  });

  await loadProject(allProjects[0]);
}

async function createDefaultProject() {
  const { data } = await sb.from('projects').insert({
    name: 'My Project', description: 'Default project', color: '#7090f0', owner_id: currentUser.id
  }).select().single();
  if (data) {
    await sb.from('project_members').insert({ project_id: data.id, user_id: currentUser.id, permission: 'admin' });
  }
}

async function loadTasks() {
  if (!currentProjectId) return;
  const { data, error } = await sb.from('tasks_with_details').select('*').eq('project_id', currentProjectId);

function daysFrom(dateStr) {
  const due = new Date(dateStr); due.setHours(0,0,0,0);
  const now = new Date(); now.setHours(0,0,0,0);
  return Math.round((due - now) / 86400000);
}
function fmtDate(dateStr) {
  return new Date(dateStr).toLocaleDateString('en-GB',{day:'2-digit',month:'short',year:'numeric'});
}
function relDate(days, status) {
  if (status === 'done') return '✓ Completed';
  if (days < 0) return `${Math.abs(days)}d overdue`;
  if (days === 0) return 'Due today!';
  if (days === 1) return '1 day left';
  return `${days} days left`;
}
function effectiveStatus(t) {
  if (t.status === 'done') return 'done';
  if (daysFrom(t.due_date) < 0) return 'overdue';
  return t.status;
}

async function saveTask() {
  const id = document.getElementById('f-task-id').value;
  const title = document.getElementById('f-title').value.trim();
  const due = document.getElementById('f-due').value;
  if (!title || !due) return notify('Title and due date required', 'warn');

  setLoading('save-task-btn', true);
  const taskData = {
    project_id: currentProjectId,
    title, due_date: due,
    description: document.getElementById('f-desc').value.trim() || null,
    priority: document.getElementById('f-priority').value,
    status: document.getElementById('f-status').value,
    tag: selectedTag,
    estimated_hours: parseFloat(document.getElementById('f-hours').value) || null,
    created_by: currentUser.id,
  };

  let taskId = id;
  if (id) {
    const { error } = await sb.from('tasks').update(taskData).eq('id', id);
    if (error) { setLoading('save-task-btn', false); return notify('Update failed: ' + error.message, 'danger'); }
  } else {
    const { data, error } = await sb.from('tasks').insert(taskData).select().single();
    if (error) { setLoading('save-task-btn', false); return notify('Create failed: ' + error.message, 'danger'); }
    taskId = data.id;
  }

  // Assignees
  await sb.from('task_assignees').delete().eq('task_id', taskId);
  const assigneeIds = Array.from(document.getElementById('f-assignees').selectedOptions).map(o => o.value);
  if (assigneeIds.length) {
    await sb.from('task_assignees').insert(assigneeIds.map(uid => ({ task_id: taskId, user_id: uid, assigned_by: currentUser.id })));
  }

  // Upload pending files
  for (const pf of pendingTaskFiles) {
    await uploadFile(pf.file, taskId, 'task');
  }
  pendingTaskFiles = [];

  // Activity log
  await logActivity(taskId, id ? 'updated task' : 'created task', title);

  setLoading('save-task-btn', false);
  closeModal('modal-task');
  await loadTasks();
  notify(id ? 'Task updated!' : 'Task created!', 'ok');
}

async function toggleDone(taskId, currentStatus) {

  await sb.from('tasks').update({ status: newStatus, completed_at: newStatus === 'done' ? new Date().toISOString() : null }).eq('id', taskId);
  await logActivity(taskId, newStatus === 'done' ? 'completed task' : 'reopened task');
  await loadTasks();
}

async function downloadFile(fileId, storagePath, fileName) {
  const { data: { session } } = await sb.auth.getSession();
  const res = await fetch(`${WORKER_URL}/file/${encodeURIComponent(storagePath)}`, {
    headers: { 'Authorization': `Bearer ${session.access_token}` }
  });
  if (!res.ok) return notify('Download failed', 'danger');
  const blob = await res.blob();
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = fileName;
  a.click();
  URL.revokeObjectURL(a.href);
}

// ─── FILE UPLOADS ─────────────────────────────────────────────────
function handleDragOver(e, zoneId) { e.preventDefault(); document.getElementById(zoneId).classList.add('dragging'); }
function handleDragLeave(zoneId) { document.getElementById(zoneId).classList.remove('dragging'); }
function handleFileDrop(e, listId, category) {
  e.preventDefault();
  handleDragLeave(e.currentTarget.id);
  addFilesToPending(Array.from(e.dataTransfer.files), listId, category);
}
function handleFileSelect(e, listId, category) {
  addFilesToPending(Array.from(e.target.files), listId, category);
  e.target.value = '';
}

function addFilesToPending(files, listId, category) {
  const allowed = ['application/pdf','application/msword',
    'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
    'application/vnd.ms-excel','application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
    'image/jpeg','image/png','image/gif','image/webp','text/plain'];
  const MAX = 50*1024*1024;
  files.forEach(file => {
    if (!allowed.includes(file.type)) return notify(`${file.name}: file type not allowed`, 'warn');
    if (file.size > MAX) return notify(`${file.name}: too large (max 50MB)`, 'warn');
    if (category === 'task') pendingTaskFiles.push({ file });
    else pendingCompleteFiles.push({ file });
    addFileItemToList(file, listId, category);
  });
}

function addFileItemToList(file, listId, category) {
  const list = document.getElementById(listId);
  const div = document.createElement('div');
  div.className = 'file-item';
  div.innerHTML = `<div class="file-icon">${fileIcon(file.name)}</div>
    <div class="file-info"><div class="file-name">${file.name}</div><div class="file-size">${fmtSize(file.size)} · Pending upload</div></div>
    <button class="file-remove" onclick="removePendingFile(this,'${file.name}','${category}')">✕</button>`;
  list.appendChild(div);
}

function removePendingFile(btn, name, category) {
  btn.closest('.file-item').remove();
  const arr = category === 'task' ? pendingTaskFiles : pendingCompleteFiles;
  const i = arr.findIndex(pf => pf.file.name === name);
  if (i >= 0) arr.splice(i, 1);
}

async function uploadFile(file, taskId, category) {
  const { data: { session } } = await sb.auth.getSession();
  if (!session) return;

  const formData = new FormData();
  formData.append('file', file);
  formData.append('task_id', taskId);
  formData.append('category', category);

  const res = await fetch(`${WORKER_URL}/upload`, {
    method: 'POST',
    headers: { 'Authorization': `Bearer ${session.access_token}` },
    body: formData,
  });

  if (!res.ok) {
    const err = await res.json().catch(() => ({}));
    notify(`Upload failed: ${err.error || res.status}`, 'danger');
    return;
  }
  const result = await res.json();

  // Save file record to Supabase
  await sb.from('task_files').insert({
    task_id: taskId,
    file_name: result.file_name,
    file_size: result.file_size,
    file_type: result.file_type,
    storage_path: result.key,
    file_category: category,
    uploaded_by: currentUser.id,
  });
}

    <div style="display:flex;align-items:center;gap:10px">
      <div class="avatar" style="background:${m.avatar_color||'#4060c0'}">${initials(m.full_name)}</div>

async function sendInvite() {
  const email = document.getElementById('invite-email').value.trim();
  const perm = document.getElementById('invite-perm').value;
  if (!email) return notify('Enter an email address', 'warn');

  // Check if user already exists in system
  const { data: existingProfile } = await sb.from('profiles').select('id').eq('email', email).single();
  if (existingProfile) {
    // Add directly
    const { error } = await sb.from('project_members').insert({
      project_id: currentProjectId, user_id: existingProfile.id, permission: perm, invited_by: currentUser.id
    });
    if (error && error.code !== '23505') return notify('Error: ' + error.message, 'danger');
    await loadMembers();
    notify(`${email} added to project!`, 'ok');
  } else {
    // Save invitation (they can accept when they sign up)
    const { error } = await sb.from('invitations').insert({
      project_id: currentProjectId, email, permission: perm, invited_by: currentUser.id
    });
    if (error) return notify('Error: ' + error.message, 'danger');
    notify(`Invitation saved for ${email}. Share the app link and ask them to sign up!`, 'ok');
  }
  document.getElementById('invite-email').value = '';
}

// ─── REALTIME ─────────────────────────────────────────────────────
function setupRealtime() {
  if (!currentProjectId) return;
  sb.channel('tasks-changes')
    .on('postgres_changes', { event: '*', schema: 'public', table: 'tasks', filter: `project_id=eq.${currentProjectId}` },
      async () => { await loadTasks(); })
    .on('postgres_changes', { event: '*', schema: 'public', table: 'comments' },
      async (payload) => {
        if (currentTaskId && payload.new?.task_id === currentTaskId) await loadComments(currentTaskId);
      })
    .subscribe();
}

// ─── HELPERS ──────────────────────────────────────────────────────
function initials(name) {
  if (!name) return '?';
  return name.split(' ').filter(Boolean).slice(0,2).map(n=>n[0]).join('').toUpperCase();
}
function fmtSize(bytes) {
  if (!bytes) return '0 B';
  if (bytes < 1024) return bytes + ' B';
  if (bytes < 1048576) return (bytes/1024).toFixed(1) + ' KB';
  return (bytes/1048576).toFixed(1) + ' MB';
}
function fileIcon(name) {
  const ext = (name||'').split('.').pop().toLowerCase();
  const m = {pdf:'📄',doc:'📝',docx:'📝',xls:'📊',xlsx:'📊',ppt:'📽',pptx:'📽',png:'🖼',jpg:'🖼',jpeg:'🖼',gif:'🖼',webp:'🖼',txt:'📃'};
  return m[ext] || '📁';
}
function timeAgo(ts) {
  const diff = (Date.now() - new Date(ts)) / 1000;
  if (diff < 60) return 'just now';
  if (diff < 3600) return Math.floor(diff/60) + 'm ago';
  if (diff < 86400) return Math.floor(diff/3600) + 'h ago';
  return Math.floor(diff/86400) + 'd ago';
}
function actionIcon(action) {
  if (action?.includes('created')) return '➕';
  if (action?.includes('completed')) return '✅';
  if (action?.includes('updated')) return '✏️';
  if (action?.includes('deleted')) return '🗑';
  if (action?.includes('comment')) return '💬';
  if (action?.includes('upload')) return '📎';
  return '📋';
}

}
